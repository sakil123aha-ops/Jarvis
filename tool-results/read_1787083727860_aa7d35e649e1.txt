    23→    23→export function TelemetryPanel({ onClose }: { onClose: () => void }) {
    24→    24→  const { containerStyle, handlers } = useDraggable({ defaultX: 16, defaultY: 16 });
    25→    25→  const [clock, setClock] = useState("");

    28→    28→  const [mem, setMem] = useState<string>("N/A");

    31→    31→  const [screenInfo, setScreenInfo] = useState("");
    32→    32→
    33→    33→  // Real clock

    35→    35→
    36→    36→  // Real FPS
    37→    37→  useEffect(() => { let frames = 0, last = performance.now(); const m = () => { frames++; const n = performance.now(); if (n - last >= 1000) { setFps(frames); frames = 0; last = n; } requestAnimationFrame(m); }; requestAnimationFrame(m); }, []);
    38→    38→
    39→    39→  // Real uptime
    40→    40→  useEffect(() => { const start = Date.now(); const id = setInterval(() => { const s = Math.floor((Date.now() - start) / 1000); const h = Math.floor(s / 3600); const m = Math.floor((s % 3600) / 60); const ss = s % 60; setUptime(s); }, 1000); return () => clearInterval(id); }, []);
    41→    41→
    42→    42→  // Real memory (performance.memory - Chrome only)
    43→    43→  useEffect(() => {
    44→    44→    const update = () => {
    45→    45→      const nav = navigator as Record<string, unknown>;
    46→    46→      const pm = nav.deviceMemory as number | undefined;
    47→    47→      if (pm) { setMem(`${pm}GB`); return; }
    48→    48→      const perf = performance as Record<string, unknown>;
    49→    49→      const memObj = perf.memory as { usedJSHeapSize: number; totalJSHeapSize: number; jsHeapSizeLimit: number } | undefined;
    50→    50→      if (memObj) { setMem(`${Math.round(memObj.usedJSHeapSize / 1048576)}/${Math.round(memObj.jsHeapSizeLimit / 1048576)}MB`); return; }
    51→    51→    };
    52→    52→    update();
    53→    53→    const id = setInterval(update, 3000);
    54→    54→    return () => clearInterval(id);

    59→    59→    const nav = navigator as Record<string, unknown>;
    60→    60→    const bat = nav.getBattery as (() => Promise<{ level: number; charging: boolean; addEventListener: (e: string, fn: () => void) => void }>) | undefined;
    61→    61→    if (!bat) return;
    62→    62→    bat().then((b) => {
    63→    63→      const upd = () => setBatt({ pct: Math.round(b.level * 100), chg: b.charging });
    64→    64→      upd();
    65→    65→      b.addEventListener("chargingchange", upd);
    66→    66→      b.addEventListener("levelchange", upd);
    67→    67→    }).catch(() => {});
    68→    68→  }, []);
    69→    69→
    70→    70→  // Real connection
    71→    71→  useEffect(() => {
    72→    72→    const upd = () => {
    73→    73→      const nav = navigator as Record<string, unknown>;
    74→    74→      const c = nav.connection as { effectiveType?: string; downlink?: number; rtt?: number } | undefined;
    75→    75→      if (c) {
    76→    76→        const type = c.effectiveType?.toUpperCase() || "--";
    77→    77→        const dl = c.downlink ? ` ${c.downlink}Mb/s` : "";
    78→    78→        const rtt = c.rtt ? ` ${c.rtt}ms` : "";
    79→    79→        setConn(`${type}${dl}${rtt}`);
    80→    80→      } else { setConn("N/A"); }
    81→    81→    };
    82→    82→    upd();
    83→    83→    window.addEventListener("online", upd);
    84→    84→    window.addEventListener("offline", upd);
    85→    85→    return () => { window.removeEventListener("online", upd); window.removeEventListener("offline", upd); };
    86→    86→  }, []);
    87→    87→
    88→    88→  // Screen info
    89→    89→  useEffect(() => { setScreenInfo(`${screen.width}x${screen.height} @${window.devicePixelRatio}x`); }, []);
    90→    90→
    91→    91→  const fmtUptime = (s: number) => { const h = Math.floor(s / 3600); const m = Math.floor((s % 3600) / 60); const ss = s % 60; return `${String(h).padStart(2, "0")}:${String(m).padStart(2, "0")}:${String(ss).padStart(2, "0")}`; };
    92→    92→
    93→    93→  return (
    94→    94→    <div className="jarvis-panel p-3 w-60 z-20" style={containerStyle}>
    95→    95→      <CornerBrackets />
    96→    96→      <PanelHeader title="SYSTEM" handlers={handlers} onClose={onClose} icon={<Cpu size={12} />} extra={<span className={navigator.onLine ? "jarvis-blink jarvis-green" : "jarvis-red"}><span>■</span> {navigator.onLine ? "ONLINE" : "OFFLINE"}</span>} />

    98→    98→        <Row label="STATE" value={fps > 0 ? "ACTIVE" : "LOADING"} accent="green" />
    99→    99→        <Row label="CLOCK" value={clock} />
   100→   100→        <Row label="FPS" value={String(fps)} accent={fps < 30 ? "amber" : fps > 0 ? "green" : undefined} />
   101→   101→        <Row label="MEMORY" value={mem} />
   102→   102→        {batt && <Row label="BATTERY" value={`${batt.pct}%${batt.chg ? " CHR" : ""}`} accent={batt.pct < 20 ? "red" : batt.chg ? "green" : undefined} />}
   103→   103→        <Row label="NETWORK" value={conn} />
   104→   104→        <Row label="SCREEN" value={screenInfo} />
   105→   105→        <Row label="UPTIME" value={fmtUptime(uptime)} />
   106→   106→        <Row label="UA" value={navigator.userAgent.includes("Chrome") ? "CHROME" : navigator.userAgent.includes("Firefox") ? "FIREFOX" : navigator.userAgent.includes("Safari") ? "SAFARI" : "BROWSER"} />

   179→   179→interface ChatMsg { id: string; role: "user" | "assistant" | "system"; ts: number; content: string; provider?: string; }

   181→   181→export function TranscriptLog({ onClose }: { onClose: () => void }) {
   182→   182→  const { containerStyle, handlers } = useDraggable({ defaultFromRight: 336, defaultY: 128 });
   183→   183→  const [messages, setMessages] = useState<ChatMsg[]>([
   184→   184→    { id: "sys", role: "system", ts: Date.now(), content: "JARVIS online. Configure an API key in Settings > Keys to enable AI chat. Supports DeepSeek, OpenAI, and Gemini." },
   185→   185→  ]);
   186→   186→  const [input, setInput] = useState("");
   187→   187→  const [loading, setLoading] = useState(false);
   188→   188→  const scrollRef = useRef<HTMLDivElement>(null);
   189→   189→
   190→   190→  useEffect(() => { if (scrollRef.current) scrollRef.current.scrollTop = scrollRef.current.scrollHeight; }, [messages]);
   191→   191→
   192→   192→  const send = async () => {
   193→   193→    const text = input.trim();
   194→   194→    if (!text || loading) return;
   195→   195→    const userMsg: ChatMsg = { id: Date.now().toString(), role: "user", ts: Date.now(), content: text };
   196→   196→    const newMsgs = [...messages.filter((m) => m.role !== "system"), userMsg];
   197→   197→    setMessages((prev) => [...prev.filter((m) => m.role !== "system"), userMsg]);
   198→   198→    setInput("");
   199→   199→    setLoading(true);
   200→   200→    try {
   201→   201→      const apiMsgs = newMsgs.map((m) => ({ role: m.role, content: m.content }));
   202→   202→      const res = await fetch("/api/chat", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ messages: apiMsgs }) });
   203→   203→      const data = await res.json();
   204→   204→      if (data.error) {
   205→   205→        setMessages((prev) => [...prev, { id: (Date.now() + 1).toString(), role: "system", ts: Date.now(), content: data.error }]);
   206→   206→      } else {
   207→   207→        setMessages((prev) => [...prev, { id: (Date.now() + 1).toString(), role: "assistant", ts: Date.now(), content: data.reply, provider: data.provider }]);
   208→   208→      }
   209→   209→    } catch {
   210→   210→      setMessages((prev) => [...prev, { id: (Date.now() + 1).toString(), role: "system", ts: Date.now(), content: "Network error. Check connection and try again." }]);
   211→   211→    }
   212→   212→    setLoading(false);
   213→   213→  };
   214→   214→
   215→   215→  return (
   216→   216→    <div className="jarvis-panel p-0 flex flex-col z-20" style={{ ...containerStyle, width: 340, height: "calc(100vh - 224px)" }}>
   217→   217→      <PanelHeader title="CONVERSATION" handlers={handlers} onClose={onClose} extra={<span className="jarvis-readout-dim text-[10px]">{messages.length} msgs</span>} />
   218→   218→      <div ref={scrollRef} className="flex-1 overflow-y-auto jarvis-scroll px-3 py-2 space-y-2 text-xs">
   219→   219→        {messages.map((t) => (
   220→   220→          <div key={t.id} className="leading-relaxed">
   221→   221→            <div className="flex items-center gap-2 text-[10px] mb-0.5">
   222→   222→              <span className={t.role === "user" ? "jarvis-amber" : t.role === "assistant" ? "jarvis-green" : "jarvis-readout-dim"}>{t.role === "user" ? "USR" : t.role === "assistant" ? "AI" : "SYS"}</span>
   223→   223→              <span className="jarvis-readout-dim">{new Date(t.ts).toISOString().slice(11, 19)}</span>
   224→   224→              {t.provider && <span className="jarvis-primary">[{t.provider}]</span>}
   225→   225→            </div>
   226→   226→            <div className="jarvis-readout whitespace-pre-wrap">{t.content}</div>
   227→   227→          </div>
   228→   228→        ))}
   229→   229→        {loading && <div className="flex items-center gap-2 jarvis-amber"><Loader2 size={12} className="animate-spin" /> Thinking...</div>}
   230→   230→      </div>
   231→   231→      <div className="px-3 py-2 border-t border-[color:var(--jarvis-panel-border)] flex gap-1">
   232→   232→        <input
   233→   233→          className="jarvis-key-input flex-1"
   234→   234→          placeholder="Type a message..."
   235→   235→          value={input}
   236→   236→          onChange={(e) => setInput(e.target.value)}

   238→   238→          disabled={loading}
   239→   239→        />

   241→   241→          <Send size={12} />

   255→   255→  useEffect(() => {
   256→   256→    let frames = 0, last = performance.now();

   258→   258→    requestAnimationFrame(m);
   259→   259→  }, []);
   260→   260→
   261→   261→  useEffect(() => {
   262→   262→    const addLog = (kind: string, ...args: unknown[]) => {
   263→   263→      const msg = args.map((a) => (typeof a === "object" ? JSON.stringify(a)?.slice(0, 120) : String(a))).join(" ");
   264→   264→      setLogs((prev) => [...prev.slice(-maxLogs), { id: Date.now().toString() + Math.random(), ts: Date.now(), kind, msg, ms: 0 }]);

   267→   267→    const origWarn = console.warn;
   268→   268→    const origErr = console.error;
   269→   269→    console.log = (...a) => { origLog(...a); addLog("info", ...a); };
   270→   270→    console.warn = (...a) => { origWarn(...a); addLog("warn", ...a); };
   271→   271→    console.error = (...a) => { origErr(...a); addLog("error", ...a); };

   274→   274→    addLog("info", `Screen: ${screen.width}x${screen.height}, DPR: ${window.devicePixelRatio}`);
   275→   275→    addLog("info", `Cores: ${navigator.hardwareConcurrency || "?"}, Memory: ${(navigator as Record<string, unknown>).deviceMemory || "?"}`);

   286→   286→        <div>ERRS <span className={logs.filter((l) => l.kind === "error").length > 0 ? "jarvis-red" : ""}>{logs.filter((l) => l.kind === "error").length}</span></div>
   287→   287→        <div>BUFFER <span>{logs.length}/{maxLogs}</span></div>
   288→   288→      </div>
   289→   289→      <div className="flex-1 overflow-y-auto jarvis-scroll px-3 py-2 space-y-1 text-[11px] font-mono">
   290→   290→        {logs.map((e) => (

   292→   292→            <span className="jarvis-readout-dim">[{new Date(e.ts).toISOString().slice(11, 23)}]</span>{" "}
   293→   293→            <span className={e.kind === "error" ? "jarvis-red" : e.kind === "warn" ? "jarvis-amber" : e.kind === "system" ? "jarvis-primary" : "jarvis-green"}>{e.kind.padEnd(7)}</span>{" "}
   294→   294→            <span className="jarvis-readout">{e.msg}</span>

   300→   300→        <button className="jarvis-btn" onClick={() => setLogs([])}>CLEAR</button>

   303→   303→    </div>
   304→   304→  );
